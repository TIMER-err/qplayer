cmake_minimum_required(VERSION 3.20)
project(qplayer_webview LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Only JNI headers are needed. Do not find/link the JVM or AWT libraries.
if(NOT DEFINED ENV{JAVA_HOME})
    message(FATAL_ERROR "JAVA_HOME must point to the build JDK")
endif()
add_library(qplayerwebview SHARED qplayer_webview.cpp)
target_include_directories(qplayerwebview PRIVATE "$ENV{JAVA_HOME}/include" upstream/src_c)
if(WIN32)
    target_include_directories(qplayerwebview PRIVATE "$ENV{JAVA_HOME}/include/win32")
    set(WEBVIEW2_SDK "$ENV{WEBVIEW2_SDK}" CACHE PATH "Unpacked Microsoft.Web.WebView2 1.0.2592.51 NuGet package")
    if(NOT EXISTS "${WEBVIEW2_SDK}/build/native/include/WebView2.h")
        message(FATAL_ERROR "Set WEBVIEW2_SDK to an unpacked Microsoft.Web.WebView2 1.0.2592.51 package")
    endif()
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "ARM64|arm64|aarch64")
        set(WV2_ARCH arm64)
    else()
        set(WV2_ARCH x64)
    endif()
    target_include_directories(qplayerwebview PRIVATE "${WEBVIEW2_SDK}/build/native/include")
    target_compile_options(qplayerwebview PRIVATE /EHsc)
    target_link_libraries(qplayerwebview PRIVATE "${WEBVIEW2_SDK}/build/native/${WV2_ARCH}/WebView2LoaderStatic.lib" advapi32 ole32 oleaut32 shlwapi shell32 user32 version winhttp)
elseif(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(qplayer_webview.cpp PROPERTIES LANGUAGE OBJCXX)
    target_include_directories(qplayerwebview PRIVATE "$ENV{JAVA_HOME}/include/darwin")
    target_compile_definitions(qplayerwebview PRIVATE WEBVIEW_COCOA=1 OBJC_OLD_DISPATCH_PROTOTYPES=1)
    target_compile_options(qplayerwebview PRIVATE -fblocks)
    target_link_libraries(qplayerwebview PRIVATE "-framework WebKit" "-framework Cocoa" "-framework QuartzCore" "-framework Security")
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
    pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11)
    pkg_check_modules(WEBKIT QUIET webkit2gtk-4.1)
    if(NOT WEBKIT_FOUND)
        pkg_check_modules(WEBKIT REQUIRED webkit2gtk-4.0)
    endif()
    target_sources(qplayerwebview PRIVATE upstream/src_c/webkit_loader.cpp)
    target_include_directories(qplayerwebview PRIVATE "$ENV{JAVA_HOME}/include/linux" ${WEBKIT_INCLUDE_DIRS})
    target_compile_definitions(qplayerwebview PRIVATE WEBVIEW_GTK=1)
    target_link_libraries(qplayerwebview PRIVATE PkgConfig::GTK PkgConfig::X11 dl pthread)
endif()
install(TARGETS qplayerwebview RUNTIME DESTINATION native-webview LIBRARY DESTINATION native-webview)
install(FILES jni-config.json resource-config.json DESTINATION META-INF/native-image/dev.t1m3.qplayer/webview)
install(FILES NOTICE upstream/LICENSE DESTINATION META-INF/licenses/qplayer-webview)
